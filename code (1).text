/* ========================================================================== *
 * 1. WEB APP & REST API ENTRY POINTS
 * ========================================================================== */

function doGet(e) {
  const action = e && e.parameter && e.parameter.action;
  if (action) {
    try {
      if (action === 'getBootstrap') {
        return jsonResponse_({ ok: true, data: getBootstrapData() });
      }
      if (action === 'ping') {
        return jsonResponse_({ ok: true, time: nowIso_(), sheetId: CONFIG.SPREADSHEET_ID });
      }
      return jsonResponse_({ ok: false, error: 'Unknown action: ' + action });
    } catch (err) {
      return jsonResponse_({ ok: false, error: err.message });
    }
  }

  // Serve index.html when opened directly in browser
  try {
    return HtmlService.createTemplateFromFile('index')
      .evaluate()
      .setTitle('ระบบบันทึกสัตว์น้ำ หว้ากอ — Waghor AquaLog')
      .addMetaTag('viewport', 'width=device-width, initial-scale=1.0, maximum-scale=5.0')
      .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
  } catch (err) {
    return ContentService.createTextOutput(
      'Waghor AquaLog API is running. Add index.html or call ?action=getBootstrap'
    );
  }
}

function doPost(e) {
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const payload = JSON.parse(raw);
    const action = payload.action;
    const clientTxId = String(payload.client_tx_id || '').trim();
    const cacheKey = clientTxId ? ('tx_' + clientTxId.slice(0, 180)) : '';

    return withLock_(() => {
      if (cacheKey) {
        try {
          const cached = CacheService.getScriptCache().get(cacheKey);
          if (cached) {
            return ContentService
              .createTextOutput(cached)
              .setMimeType(ContentService.MimeType.JSON);
          }
        } catch (cacheErr) {}
      }

      let result;
      switch (action) {
        case 'recordDeath':
          result = recordDeath(payload);
          break;
        case 'recordMovement':
          result = recordMovement(payload);
          break;
        case 'createTransfer':
          result = createTransfer(payload);
          break;
        case 'confirmTransfer':
          result = confirmTransfer(payload.move_id, payload.received_by);
          break;
        case 'rejectTransfer':
          result = rejectTransfer(payload.move_id, payload.received_by);
          break;
        case 'lockMonth':
          result = lockMonth(payload.period_ym, payload.locked_by);
          break;
        case 'addStaff':
          result = addStaff(payload);
          break;
        case 'addSpecies':
          result = addSpecies(payload);
          break;
        case 'assignTankOwner':
          result = assignTankOwner(payload.tank_id, payload.owner_staff_id);
          break;
        case 'editMortalityDetails':
          result = editMortalityDetails(payload);
          break;
        case 'updateLot':
          result = updateLot(payload);
          break;
        case 'cancelMortality':
          result = cancelMortality(payload.death_id, payload.cancelled_by);
          break;
        case 'cancelMovement':
          result = cancelMovement(payload.move_id, payload.cancelled_by);
          break;
        case 'syncCentralSpecies':
          result = syncCentralSpecies();
          break;
        default:
          throw new Error('ไม่รองรับคำสั่ง: ' + action);
      }

      const responseObj = { ok: true, data: result };
      const serialized = JSON.stringify(responseObj);
      if (cacheKey) {
        try {
          CacheService.getScriptCache().put(cacheKey, serialized, 21600);
        } catch (cacheErr) {}
      }
      return ContentService
        .createTextOutput(serialized)
        .setMimeType(ContentService.MimeType.JSON);
    });
  } catch (err) {
    return jsonResponse_({ ok: false, error: err.message });
  }
}

/**
 * บันทึกซากสัตว์น้ำ (ตัดสต็อกแบบ FIFO อัตโนมัติ + ตรวจบังคับชั่งน้ำหนักปลาใหญ่ + ตรวจล็อกเดือน)
 */
function recordDeath(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const deathDate = String(payload.death_date || todayIso_());
    assertMonthNotLocked_(ss, deathDate);

    const tankId = String(payload.tank_id || '').trim();
    const speciesId = String(payload.species_id || '').trim();
    const disposalType = String(payload.disposal_type || 'เก็บซาก').trim();
    const qty = Number(payload.qty || 0);
    const symptom = String(payload.symptom || '').trim();
    const sizeCm = payload.size_cm !== '' && payload.size_cm != null ? Number(payload.size_cm) : '';
    const weightG = payload.weight_g !== '' && payload.weight_g != null ? Number(payload.weight_g) : '';
    const sourceNote = String(payload.source_note || '').trim();
    const recordedBy = String(payload.recorded_by || 'STF001').trim();

    if (!tankId || !speciesId) throw new Error('กรุณาระบุตู้และชนิดสัตว์น้ำ');
    if (qty < 1) throw new Error('จำนวนต้องมากกว่าหรือเท่ากับ 1 ตัว');
    if (!symptom) throw new Error('กรุณาระบุอาการที่พบ (บังคับทุกเคส)');

    const sp = findSpecies_(ss, speciesId);
    if (!sp) throw new Error('ไม่พบรหัสชนิดสัตว์น้ำ: ' + speciesId);

    const largeIds = readObjects_(ss.getSheetByName(CONFIG.SHEETS.RULES))
      .filter(r => toBool_(r.require_weight_on_death))
      .map(r => String(r.species_id));
    const requireWeight = largeIds.indexOf(speciesId) !== -1 || sp.pricing_unit === 'PER_KG';
    if (requireWeight && (!weightG || weightG <= 0)) {
      throw new Error('สัตว์น้ำขนาดใหญ่หรือคิดราคาต่อ กก. บังคับชั่งน้ำหนัก (กรัม) ตอนตาย');
    }

    const targetLotId = String(payload.target_lot_id || '').trim();
    const fifo = deductFifoLots_(ss, tankId, speciesId, qty, weightG, targetLotId, deathDate);

    const shMort = ss.getSheetByName(CONFIG.SHEETS.MORTALITY);
    const deathId = makeId_('DTH', deathDate, shMort, 'death_id');
    const unitCostApplied = fifo.breakdown.length ? fifo.breakdown[0].unit_cost : 0;
    const warrantyPrefix = fifo.hasWarranty ? '[ตายในประกัน - เคลมร้าน] ' : '';
    const noteBase = warrantyPrefix + (sourceNote || (fifo.breakdown.length ? fifo.breakdown[0].source_type : ''));
    const noteText = noteBase + ' | LOTS:' + JSON.stringify(fifo.breakdown);

    shMort.appendRow([
      deathId,
      deathDate,
      tankId,
      speciesId,
      sp.species_name_th,
      disposalType,
      qty,
      symptom,
      sizeCm,
      weightG,
      unitCostApplied,
      fifo.totalCost,
      noteText,
      recordedBy
    ]);

    return { death_id: deathId, total_loss_value: fifo.totalCost, fifo_breakdown: fifo.breakdown };
  });
}

/**
 * บันทึกรับเข้า / ย้ายออกนอกตู้จัดแสดง / งานสุขภาพ (ส่ง-รับคืน)
 */
function recordMovement(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const moveDate = String(payload.move_date || todayIso_());
    assertMonthNotLocked_(ss, moveDate);

    const direction = String(payload.direction || 'IN').trim();
    const category = String(payload.category || '').trim();
    const tankId = String(payload.tank_id || '').trim();
    const speciesId = String(payload.species_id || '').trim();
    const qty = Number(payload.qty || 0);
    const note = String(payload.note || '').trim();
    const createdBy = String(payload.created_by || 'STF001').trim();

    if (!tankId || !speciesId || qty < 1) throw new Error('กรุณากรอกข้อมูลตู้ ชนิด และจำนวนให้ครบ');

    const sp = findSpecies_(ss, speciesId);
    if (!sp) throw new Error('ไม่พบรหัสชนิดสัตว์น้ำ: ' + speciesId);
    const tank = findTank_(ss, tankId);
    const waterType = tank ? tank.water_type : sp.water_type;

    const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
    const moveId = makeId_('MOV', moveDate, shMove, 'move_id');
    const isIntake = (direction === 'IN' || direction === 'HEALTH_RETURN');

    if (isIntake) {
      const pricingUnit = String(payload.pricing_unit || sp.pricing_unit || 'PER_UNIT');
      const unitCost = Math.max(0, Number(payload.unit_cost || 0));
      const warrantyUntil = String(payload.warranty_until || '').trim();
      const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
      const lotId = makeId_('LOT', moveDate, shLots, 'lot_id');
      const lotHeaders = shLots.getLastRow() >= 1
        ? shLots.getRange(1, 1, 1, shLots.getLastColumn()).getValues()[0].map(h => String(h).trim())
        : [];
      const lotMap = {
        lot_id: lotId, tank_id: tankId, species_id: speciesId,
        species_name_snapshot: sp.species_name_th, water_type: waterType,
        received_date: moveDate, source_type: category || 'รับเข้า',
        pricing_unit: pricingUnit, unit_cost: unitCost,
        qty_initial: qty, qty_remaining: qty, status: 'ACTIVE',
        warranty_until: warrantyUntil
      };
      if (lotHeaders.length >= 12) {
        shLots.appendRow(lotHeaders.map(h => Object.prototype.hasOwnProperty.call(lotMap, h) ? lotMap[h] : ''));
      } else {
        shLots.appendRow([
          lotId, tankId, speciesId, sp.species_name_th, waterType,
          moveDate, category || 'รับเข้า', pricingUnit, unitCost, qty, qty, 'ACTIVE', warrantyUntil
        ]);
      }

      shMove.appendRow([
        moveId, moveDate, direction, category, '', tankId, speciesId, sp.species_name_th,
        qty, pricingUnit, unitCost, 'COMPLETED', createdBy, createdBy, moveDate, note
      ]);

      return { move_id: moveId, lot_id: lotId, warranty_until: warrantyUntil };
    } else {
      const targetLotId = String(payload.target_lot_id || '').trim();
      const fifo = deductFifoLots_(ss, tankId, speciesId, qty, 0, targetLotId, '');
      const pricingUnit = fifo.breakdown.length ? fifo.breakdown[0].pricing_unit : 'PER_UNIT';
      const unitCost = fifo.breakdown.length ? fifo.breakdown[0].unit_cost : 0;

      shMove.appendRow([
        moveId, moveDate, direction, category, tankId, category, speciesId, sp.species_name_th,
        qty, pricingUnit, unitCost, 'COMPLETED', createdBy, '', '', (note || '') + ' | LOTS:' + JSON.stringify(fifo.breakdown)
      ]);

      return { move_id: moveId, fifo_breakdown: fifo.breakdown };
    }
  });
}

/**
 * ย้ายข้ามตู้ ขั้นตอนที่ 1: เจ้าของตู้ต้นทางกดส่ง (ตัด FIFO ต้นทาง + สร้างสถานะ PENDING_RECEIVE)
 */
function createTransfer(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const moveDate = String(payload.move_date || todayIso_());
    assertMonthNotLocked_(ss, moveDate);

    const fromTankId = String(payload.from_tank_id || '').trim();
    const toTankId = String(payload.to_tank_id || '').trim();
    const speciesId = String(payload.species_id || '').trim();
    const qty = Number(payload.qty || 0);
    const note = String(payload.note || '').trim();
    const createdBy = String(payload.created_by || 'STF001').trim();

    if (!fromTankId || !toTankId || fromTankId === toTankId) {
      throw new Error('กรุณาระบุตู้ต้นทางและตู้ปลายทางให้ถูกต้อง');
    }
    if (!speciesId || qty < 1) throw new Error('กรุณาระบุชนิดและจำนวนสัตว์น้ำ');

    const sp = findSpecies_(ss, speciesId);
    if (!sp) throw new Error('ไม่พบรหัสชนิดสัตว์น้ำ: ' + speciesId);

    const targetLotId = String(payload.target_lot_id || '').trim();
    const fifo = deductFifoLots_(ss, fromTankId, speciesId, qty, 0, targetLotId, '');
    const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
    const moveId = makeId_('TRF', moveDate, shMove, 'move_id');
    const pricingUnit = fifo.breakdown.length ? fifo.breakdown[0].pricing_unit : 'PER_UNIT';
    const unitCost = fifo.breakdown.length ? fifo.breakdown[0].unit_cost : 0;

    // เก็บข้อมูลล็อตที่ย้ายไว้ใน note แบบแยกแท็กเพื่อให้โครงสร้าง 16 คอลัมน์เดิมใช้งานได้ทันที
    const carriedTag = ' | LOTS:' + JSON.stringify(fifo.breakdown);
    shMove.appendRow([
      moveId, moveDate, 'TRANSFER', 'ตู้จัดแสดง', fromTankId, toTankId,
      speciesId, sp.species_name_th, qty, pricingUnit, unitCost,
      'PENDING_RECEIVE', createdBy, '', '', (note || ('ย้ายจาก ' + fromTankId + ' ไป ' + toTankId)) + carriedTag
    ]);

    return { move_id: moveId, transfer_status: 'PENDING_RECEIVE', carried_lots: fifo.breakdown };
  });
}

/**
 * ย้ายข้ามตู้ ขั้นตอนที่ 2: เจ้าของตู้ปลายทางกดยืนยันรับเข้า (โอนล็อต + วันที่และต้นทุนเดิมเข้าตู้ปลายทาง)
 */
function confirmTransfer(moveId, receivedBy) {
  return withLock_(() => {
    const ss = getSS_();
    const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
    const rows = shMove.getDataRange().getValues();
    const headers = rows[0];
    const idx = indexMap_(headers);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][idx.move_id]) === String(moveId)) {
        if (String(rows[r][idx.transfer_status]) !== 'PENDING_RECEIVE') {
          throw new Error('รายการนี้ถูกดำเนินการไปแล้ว');
        }
        const toTankId = String(rows[r][idx.to_tank_id]);
        const speciesId = String(rows[r][idx.species_id]);
        const spName = String(rows[r][idx.species_name_snapshot]);
        const qty = Number(rows[r][idx.qty]);
        const moveDate = formatDateCell_(rows[r][idx.move_date]);
        const noteRaw = String(rows[r][idx.note] || '');
        const carried = parseCarriedLots_(noteRaw, moveDate, String(rows[r][idx.pricing_unit]), Number(rows[r][idx.unit_cost]), qty);

        const destTank = findTank_(ss, toTankId);
        const sp = findSpecies_(ss, speciesId);
        const waterType = destTank ? destTank.water_type : (sp ? sp.water_type : 'น้ำเค็ม');
        const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
        const lotHeaders = shLots.getLastRow() >= 1
          ? shLots.getRange(1, 1, 1, shLots.getLastColumn()).getValues()[0].map(h => String(h).trim())
          : [];

        carried.forEach(item => {
          const newLotId = makeId_('LOT', moveDate, shLots, 'lot_id');
          const lotMap = {
            lot_id: newLotId, tank_id: toTankId, species_id: speciesId,
            species_name_snapshot: spName, water_type: waterType,
            received_date: item.received_date, source_type: item.source_type,
            pricing_unit: item.pricing_unit, unit_cost: item.unit_cost,
            qty_initial: item.qty_deducted, qty_remaining: item.qty_deducted, status: 'ACTIVE',
            warranty_until: item.warranty_until || ''
          };
          if (lotHeaders.length >= 12) {
            shLots.appendRow(lotHeaders.map(h => Object.prototype.hasOwnProperty.call(lotMap, h) ? lotMap[h] : ''));
          } else {
            shLots.appendRow([
              newLotId, toTankId, speciesId, spName, waterType,
              item.received_date, item.source_type, item.pricing_unit, item.unit_cost,
              item.qty_deducted, item.qty_deducted, 'ACTIVE', item.warranty_until || ''
            ]);
          }
        });

        const today = todayIso_();
        shMove.getRange(r + 1, idx.transfer_status + 1).setValue('COMPLETED');
        shMove.getRange(r + 1, idx.received_by + 1).setValue(String(receivedBy || 'STF001'));
        shMove.getRange(r + 1, idx.received_at + 1).setValue(today);
        return { move_id: moveId, transfer_status: 'COMPLETED', received_at: today };
      }
    }
    throw new Error('ไม่พบรายการย้ายรหัส: ' + moveId);
  });
}

/**
 * ย้ายข้ามตู้ กรณีตีกลับ: คืนจำนวนกลับล็อตเดิมของตู้ต้นทาง
 */
function rejectTransfer(moveId, receivedBy) {
  return withLock_(() => {
    const ss = getSS_();
    const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
    const rows = shMove.getDataRange().getValues();
    const idx = indexMap_(rows[0]);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][idx.move_id]) === String(moveId)) {
        if (String(rows[r][idx.transfer_status]) !== 'PENDING_RECEIVE') {
          throw new Error('รายการนี้ถูกดำเนินการไปแล้ว');
        }
        const noteRaw = String(rows[r][idx.note] || '');
        const carried = parseCarriedLots_(noteRaw, '', 'PER_UNIT', 0, 0);

        const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
        const lotRows = shLots.getDataRange().getValues();
        const lIdx = indexMap_(lotRows[0]);

        carried.forEach(item => {
          if (!item.lot_id) return;
          for (let lr = 1; lr < lotRows.length; lr++) {
            if (String(lotRows[lr][lIdx.lot_id]) === String(item.lot_id)) {
              const cur = Number(lotRows[lr][lIdx.qty_remaining] || 0);
              shLots.getRange(lr + 1, lIdx.qty_remaining + 1).setValue(cur + Number(item.qty_deducted));
              shLots.getRange(lr + 1, lIdx.status + 1).setValue('ACTIVE');
              break;
            }
          }
        });

        const today = todayIso_();
        shMove.getRange(r + 1, idx.transfer_status + 1).setValue('REJECTED');
        shMove.getRange(r + 1, idx.received_by + 1).setValue(String(receivedBy || 'STF001'));
        shMove.getRange(r + 1, idx.received_at + 1).setValue(today);
        return { move_id: moveId, transfer_status: 'REJECTED' };
      }
    }
    throw new Error('ไม่พบรายการย้ายรหัส: ' + moveId);
  });
}

/**
 * ปิดยอดและล็อกเดือนลงแท็บ Monthly_Snapshots
 */
function lockMonth(periodYmIso, lockedBy) {
  return withLock_(() => {
    const ss = getSS_();
    const ymIso = toIsoYM_(periodYmIso);
    const ymTh = toBuddhistYM_(ymIso + '-01');
    assertMonthNotLocked_(ss, ymIso + '-01');

    const data = getBootstrapData();
    const prevSnaps = data.monthlySnapshots.filter(s => s.is_locked).sort((a, b) => b.period_ym.localeCompare(a.period_ym));
    const prevYm = prevSnaps.length ? prevSnaps[0].period_ym : '';
    const prevFw = prevSnaps.find(s => s.period_ym === prevYm && s.water_type === 'น้ำจืด');
    const prevSw = prevSnaps.find(s => s.period_ym === prevYm && s.water_type === 'น้ำเค็ม');

    function summarize(wtype, prev) {
      const tankSet = {};
      data.tanks.filter(t => t.water_type === wtype && t.tank_type === 'DISPLAY').forEach(t => { tankSet[t.tank_id] = true; });
      const mMorts = data.mortalityLogs.filter(m => m.death_date.indexOf(ymIso) === 0 && tankSet[m.tank_id]);
      const mMoves = data.movementLogs.filter(m => m.move_date.indexOf(ymIso) === 0 && m.transfer_status === 'COMPLETED');

      let qtyIn = 0, qtyOut = 0;
      mMoves.forEach(m => {
        if ((m.direction === 'IN' || m.direction === 'HEALTH_RETURN') && tankSet[m.to_tank_id]) qtyIn += m.qty;
        if ((m.direction === 'OUT' || m.direction === 'HEALTH_OUT') && tankSet[m.from_tank_id]) qtyOut += m.qty;
      });

      const kept = mMorts.filter(m => m.disposal_type === 'เก็บซาก').reduce((s, m) => s + m.qty, 0);
      const unkept = mMorts.filter(m => m.disposal_type === 'ไม่เก็บซาก').reduce((s, m) => s + m.qty, 0);
      const loss = Math.round(mMorts.reduce((s, m) => s + m.total_loss_value, 0) * 100) / 100;

      const activeLots = data.lots.filter(l => l.water_type === wtype && l.qty_remaining > 0);
      const spEndMap = {};
      activeLots.forEach(l => { spEndMap[l.species_id] = true; });
      const speciesEnd = Object.keys(spEndMap).length;
      const qtyEnd = activeLots.reduce((s, l) => s + l.qty_remaining, 0);

      return {
        species_start: prev ? prev.species_end : speciesEnd,
        qty_start: prev ? prev.qty_end : qtyEnd,
        qty_in: qtyIn,
        qty_out: qtyOut,
        qty_dead_kept: kept,
        qty_dead_unkept: unkept,
        species_end: speciesEnd,
        total_loss_baht: loss
      };
    }

    const fw = summarize('น้ำจืด', prevFw);
    const sw = summarize('น้ำเค็ม', prevSw);
    const shSnap = ss.getSheetByName(CONFIG.SHEETS.SNAPSHOTS);
    const startRow = shSnap.getLastRow() + 1;
    const rFw = startRow;
    const rSw = startRow + 1;
    const rAll = startRow + 2;
    const today = todayIso_();
    const who = String(lockedBy || 'STF001');

    shSnap.appendRow([
      ymTh, 'น้ำจืด', fw.species_start, fw.qty_start, fw.qty_in, fw.qty_out,
      fw.qty_dead_kept, fw.qty_dead_unkept, `=G${rFw}+H${rFw}`, fw.species_end,
      `=(D${rFw}+E${rFw})-(F${rFw}+I${rFw})`, `=ROUND(I${rFw}/D${rFw}*100,2)`, fw.total_loss_baht,
      true, who, today
    ]);
    shSnap.appendRow([
      ymTh, 'น้ำเค็ม', sw.species_start, sw.qty_start, sw.qty_in, sw.qty_out,
      sw.qty_dead_kept, sw.qty_dead_unkept, `=G${rSw}+H${rSw}`, sw.species_end,
      `=(D${rSw}+E${rSw})-(F${rSw}+I${rSw})`, `=ROUND(I${rSw}/D${rSw}*100,2)`, sw.total_loss_baht,
      true, who, today
    ]);
    shSnap.appendRow([
      ymTh, 'รวม', `=SUM(C${rFw}:C${rSw})`, `=SUM(D${rFw}:D${rSw})`, `=SUM(E${rFw}:E${rSw})`, `=SUM(F${rFw}:F${rSw})`,
      `=SUM(G${rFw}:G${rSw})`, `=SUM(H${rFw}:H${rSw})`, `=SUM(I${rFw}:I${rSw})`, `=SUM(J${rFw}:J${rSw})`,
      `=SUM(K${rFw}:K${rSw})`, `=ROUND(I${rAll}/D${rAll}*100,2)`, `=SUM(M${rFw}:M${rSw})`,
      true, who, today
    ]);

    return { period_ym: ymTh, is_locked: true, locked_at: today };
  });
}

/**
 * แก้ไขรายละเอียดซากสัตว์น้ำ (เฉพาะฟิลด์ที่ไม่กระทบจำนวนสต็อก และต้องอยู่ในเดือนที่ยังไม่ล็อก)
 */
function editMortalityDetails(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const deathId = String(payload.death_id || '').trim();
    const disposalType = String(payload.disposal_type || '').trim();
    const symptom = String(payload.symptom || '').trim();
    const sizeCm = payload.size_cm !== '' && payload.size_cm != null ? Number(payload.size_cm) : '';
    const sourceNote = String(payload.source_note || '').trim();

    if (!deathId) throw new Error('ไม่พบรหัสรายการซาก');
    if (!symptom) throw new Error('กรุณาระบุอาการที่พบ (บังคับทุกเคส)');

    const shMort = ss.getSheetByName(CONFIG.SHEETS.MORTALITY);
    const rows = shMort.getDataRange().getValues();
    const idx = indexMap_(rows[0]);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][idx.death_id]) === deathId) {
        const deathDate = formatDateCell_(rows[r][idx.death_date]);
        assertMonthNotLocked_(ss, deathDate);

        // Preserve any LOTS: JSON tag already in source_note
        const existingNote = String(rows[r][idx.source_note] || '');
        const tagPos = existingNote.indexOf(' | LOTS:');
        const lotsTag = tagPos !== -1 ? existingNote.slice(tagPos) : '';

        if (disposalType) shMort.getRange(r + 1, idx.disposal_type + 1).setValue(disposalType);
        shMort.getRange(r + 1, idx.symptom + 1).setValue(symptom);
        shMort.getRange(r + 1, idx.size_cm + 1).setValue(sizeCm);
        shMort.getRange(r + 1, idx.source_note + 1).setValue(sourceNote + lotsTag);
        return { death_id: deathId, updated: true };
      }
    }
    throw new Error('ไม่พบรายการซากรหัส: ' + deathId);
  });
}

/**
 * แก้ไขข้อมูลล็อต (วันรับเข้า / ราคาต่อหน่วย / แหล่งที่มา / วันหมดประกัน)
 */
function updateLot(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const lotId = String(payload.lot_id || '').trim();
    const receivedDate = String(payload.received_date || '').trim();
    const warrantyUntil = String(payload.warranty_until || '').trim();
    const hasUnitCost = payload.unit_cost !== undefined && payload.unit_cost !== null && payload.unit_cost !== '';
    const newCost = hasUnitCost ? Math.max(0, Number(payload.unit_cost)) : null;
    const newSource = String(payload.source_type || '').trim();

    if (!lotId) throw new Error('ไม่พบรหัสล็อต');
    if (!/^\d{4}-\d{2}-\d{2}$/.test(receivedDate)) {
      throw new Error('รูปแบบวันรับเข้าไม่ถูกต้อง (ต้องเป็น YYYY-MM-DD)');
    }
    if (warrantyUntil && !/^\d{4}-\d{2}-\d{2}$/.test(warrantyUntil)) {
      throw new Error('รูปแบบวันหมดประกันไม่ถูกต้อง (ต้องเป็น YYYY-MM-DD)');
    }

    const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
    const rows = shLots.getDataRange().getValues();
    const lIdx = indexMap_(rows[0]);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][lIdx.lot_id]).trim() === lotId) {
        const oldDate = formatDateCell_(rows[r][lIdx.received_date]);
        if (oldDate) assertMonthNotLocked_(ss, oldDate);
        assertMonthNotLocked_(ss, receivedDate);

        const tankId = String(rows[r][lIdx.tank_id]).trim();
        const speciesId = String(rows[r][lIdx.species_id]).trim();
        const qtyInitial = Number(rows[r][lIdx.qty_initial] || 0);

        shLots.getRange(r + 1, lIdx.received_date + 1).setValue(receivedDate);
        if (newCost !== null && !isNaN(newCost) && lIdx.unit_cost != null) {
          shLots.getRange(r + 1, lIdx.unit_cost + 1).setValue(newCost);
        }
        if (newSource && lIdx.source_type != null) {
          shLots.getRange(r + 1, lIdx.source_type + 1).setValue(newSource);
        }
        if (lIdx.warranty_until != null) {
          shLots.getRange(r + 1, lIdx.warranty_until + 1).setValue(warrantyUntil);
        } else if (warrantyUntil) {
          const nextCol = shLots.getLastColumn() + 1;
          shLots.getRange(1, nextCol).setValue('warranty_until');
          shLots.getRange(r + 1, nextCol).setValue(warrantyUntil);
        }

        // Sync matching intake row in Movement_Logs
        const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
        const mRows = shMove.getDataRange().getValues();
        const mIdx = indexMap_(mRows[0]);
        for (let mr = mRows.length - 1; mr >= 1; mr--) {
          const dir = String(mRows[mr][mIdx.direction]).trim();
          if (
            (dir === 'IN' || dir === 'HEALTH_RETURN') &&
            String(mRows[mr][mIdx.to_tank_id]).trim() === tankId &&
            String(mRows[mr][mIdx.species_id]).trim() === speciesId &&
            formatDateCell_(mRows[mr][mIdx.move_date]) === oldDate &&
            Number(mRows[mr][mIdx.qty] || 0) === qtyInitial
          ) {
            shMove.getRange(mr + 1, mIdx.move_date + 1).setValue(receivedDate);
            if (mIdx.received_at != null) shMove.getRange(mr + 1, mIdx.received_at + 1).setValue(receivedDate);
            if (newCost !== null && !isNaN(newCost) && mIdx.unit_cost != null) {
              shMove.getRange(mr + 1, mIdx.unit_cost + 1).setValue(newCost);
            }
            if (newSource && mIdx.category != null) {
              shMove.getRange(mr + 1, mIdx.category + 1).setValue(newSource);
            }
            break;
          }
        }

        return {
          lot_id: lotId,
          updated: true,
          received_date: receivedDate,
          unit_cost: newCost !== null && !isNaN(newCost) ? newCost : Number(rows[r][lIdx.unit_cost] || 0),
          source_type: newSource || String(rows[r][lIdx.source_type] || ''),
          warranty_until: warrantyUntil
        };
      }
    }
    throw new Error('ไม่พบล็อตรหัส: ' + lotId);
  });
}

/**
 * ยกเลิกรายการซากและคืนสต็อกกลับล็อต FIFO เดิม (เฉพาะเดือนที่ยังไม่ล็อก)
 */
function cancelMortality(deathId, cancelledBy) {
  return withLock_(() => {
    const ss = getSS_();
    const shMort = ss.getSheetByName(CONFIG.SHEETS.MORTALITY);
    const rows = shMort.getDataRange().getValues();
    const idx = indexMap_(rows[0]);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][idx.death_id]) === String(deathId)) {
        const deathDate = formatDateCell_(rows[r][idx.death_date]);
        assertMonthNotLocked_(ss, deathDate);

        const tankId = String(rows[r][idx.tank_id]);
        const speciesId = String(rows[r][idx.species_id]);
        const qty = Number(rows[r][idx.qty] || 0);
        const noteRaw = String(rows[r][idx.source_note] || '');
        const carried = parseCarriedLots_(noteRaw, deathDate, 'PER_UNIT', Number(rows[r][idx.unit_cost_applied] || 0), qty);

        restoreFifoLots_(ss, tankId, speciesId, carried, qty);
        shMort.deleteRow(r + 1);
        return { death_id: deathId, cancelled: true, restored_qty: qty, cancelled_by: String(cancelledBy || 'STF001') };
      }
    }
    throw new Error('ไม่พบรายการซากรหัส: ' + deathId);
  });
}

/**
 * ยกเลิกรายการรับเข้า-ย้ายออก และปรับคืนสต็อก FIFO ให้ตรงกัน (เฉพาะเดือนที่ยังไม่ล็อก)
 */
function cancelMovement(moveId, cancelledBy) {
  return withLock_(() => {
    const ss = getSS_();
    const shMove = ss.getSheetByName(CONFIG.SHEETS.MOVEMENTS);
    const rows = shMove.getDataRange().getValues();
    const idx = indexMap_(rows[0]);

    for (let r = 1; r < rows.length; r++) {
      if (String(rows[r][idx.move_id]) === String(moveId)) {
        const moveDate = formatDateCell_(rows[r][idx.move_date]);
        assertMonthNotLocked_(ss, moveDate);

        const dir = String(rows[r][idx.direction]);
        const status = String(rows[r][idx.transfer_status]);
        const fromTank = String(rows[r][idx.from_tank_id]);
        const toTank = String(rows[r][idx.to_tank_id]);
        const speciesId = String(rows[r][idx.species_id]);
        const qty = Number(rows[r][idx.qty] || 0);
        const noteRaw = String(rows[r][idx.note] || '');

        if (dir === 'TRANSFER' && status === 'PENDING_RECEIVE') {
          return rejectTransfer(moveId, cancelledBy);
        }
        if (dir === 'TRANSFER' && status === 'COMPLETED') {
          throw new Error('รายการย้ายข้ามตู้ที่ยืนยันรับเข้าแล้ว กรุณาใช้วิธีย้ายกลับตู้เดิมแทนการลบ');
        }

        const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
        const lotRows = shLots.getDataRange().getValues();
        const lIdx = indexMap_(lotRows[0]);

        if (dir === 'IN' || dir === 'HEALTH_RETURN') {
          // Find the exact lot created by this intake (search from newest row upwards)
          let foundRow = -1;
          for (let lr = lotRows.length - 1; lr >= 1; lr--) {
            if (
              String(lotRows[lr][lIdx.tank_id]) === toTank &&
              String(lotRows[lr][lIdx.species_id]) === speciesId &&
              formatDateCell_(lotRows[lr][lIdx.received_date]) === moveDate &&
              Number(lotRows[lr][lIdx.qty_initial]) === qty
            ) {
              if (Number(lotRows[lr][lIdx.qty_remaining]) < qty) {
                throw new Error('ล็อตนี้ถูกตัดสต็อกไปแล้วบางส่วน กรุณายกเลิกรายการตาย/ย้ายออกที่ใช้ล็อตนี้ก่อน');
              }
              foundRow = lr + 1;
              break;
            }
          }
          if (foundRow === -1) throw new Error('ไม่พบล็อตสต็อกที่ตรงกับรายการรับเข้านี้');
          shLots.deleteRow(foundRow);
        } else {
          const carried = parseCarriedLots_(noteRaw, moveDate, String(rows[r][idx.pricing_unit]), Number(rows[r][idx.unit_cost] || 0), qty);
          restoreFifoLots_(ss, fromTank, speciesId, carried, qty);
        }

        shMove.deleteRow(r + 1);
        return { move_id: moveId, cancelled: true, cancelled_by: String(cancelledBy || 'STF001') };
      }
    }
    throw new Error('ไม่พบรายการรับเข้า-ย้ายออกรหัส: ' + moveId);
  });
}

function restoreFifoLots_(ss, tankId, speciesId, carried, fallbackQty) {
  const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
  const lotRows = shLots.getDataRange().getValues();
  const lIdx = indexMap_(lotRows[0]);
  let restored = 0;

  carried.forEach(item => {
    if (!item.lot_id) return;
    for (let lr = 1; lr < lotRows.length; lr++) {
      if (String(lotRows[lr][lIdx.lot_id]) === String(item.lot_id)) {
        const cur = Number(lotRows[lr][lIdx.qty_remaining] || 0);
        const add = Number(item.qty_deducted || 0);
        shLots.getRange(lr + 1, lIdx.qty_remaining + 1).setValue(cur + add);
        shLots.getRange(lr + 1, lIdx.status + 1).setValue('ACTIVE');
        restored += add;
        break;
      }
    }
  });

  // Fallback for legacy rows without lot_id tag: restore to most recent lot in that tank
  if (restored === 0 && fallbackQty > 0) {
    for (let lr = lotRows.length - 1; lr >= 1; lr--) {
      if (String(lotRows[lr][lIdx.tank_id]) === tankId && String(lotRows[lr][lIdx.species_id]) === speciesId) {
        const cur = Number(lotRows[lr][lIdx.qty_remaining] || 0);
        shLots.getRange(lr + 1, lIdx.qty_remaining + 1).setValue(cur + fallbackQty);
        shLots.getRange(lr + 1, lIdx.status + 1).setValue('ACTIVE');
        break;
      }
    }
  }
}

function addStaff(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const name = String(payload.full_name || '').trim();
    const pos = String(payload.position || 'เจ้าหน้าที่ดูแลตู้').trim();
    const role = String(payload.role || 'STAFF').trim();
    if (!name) throw new Error('กรุณากรอกชื่อ-สกุล');

    const sh = ss.getSheetByName(CONFIG.SHEETS.STAFF);
    const existingStaff = readObjects_(sh);
    let maxStaffN = 0;
    existingStaff.forEach(s => {
      const m = String(s.staff_id || '').trim().match(/^STF(\d+)$/i);
      if (m) {
        const n = parseInt(m[1], 10);
        if (n > maxStaffN) maxStaffN = n;
      }
    });
    const id = 'STF' + String(maxStaffN + 1).padStart(3, '0');
    sh.appendRow([id, name, pos, role, true]);
    return { staff_id: id, full_name: name, position: pos, role };
  });
}

}

/**
 * เพิ่มชนิดสัตว์น้ำใหม่ (นอกบัญชีกลาง) ลงแท็บ Master_Species และ Species_Rules
 * ใช้รหัส SPX001, SPX002, ... เพื่อไม่ให้ชนรหัสบัญชีกลางและถูกรักษาไว้เมื่อซิงก์ชีตกลาง
 */
function addSpecies(payload) {
  return withLock_(() => {
    const ss = getSS_();
    const name = String(payload.species_name_th || '').trim();
    const waterType = String(payload.water_type || '').trim();
    const pricingUnit = String(payload.pricing_unit || 'PER_UNIT').trim();
    const refPrice = Math.max(0, Number(payload.reference_price || 0));
    const requireWeight = toBool_(payload.require_weight_on_death) || pricingUnit === 'PER_KG';

    if (!name) throw new Error('กรุณาระบุชื่อชนิดสัตว์น้ำ');
    if (waterType !== 'น้ำจืด' && waterType !== 'น้ำเค็ม') {
      throw new Error('กรุณาเลือกประเภทน้ำ (น้ำจืด หรือ น้ำเค็ม)');
    }
    if (pricingUnit !== 'PER_UNIT' && pricingUnit !== 'PER_KG') {
      throw new Error('หน่วยคิดราคาไม่ถูกต้อง');
    }

    const shSpecies = ss.getSheetByName(CONFIG.SHEETS.SPECIES);
    if (!shSpecies) throw new Error('ไม่พบแท็บ ' + CONFIG.SHEETS.SPECIES);

    const existingList = readObjects_(shSpecies);
    const norm = s => String(s || '').replace(/\s+/g, '').toLowerCase();
    const normTarget = norm(name);
    const dup = existingList.find(s => norm(s.species_name_th) === normTarget);
    if (dup) {
      throw new Error('มีชนิดสัตว์น้ำ "' + name + '" อยู่ในบัญชีแล้ว (' + String(dup.species_id || '').trim() + ')');
    }

    let maxN = 0;
    existingList.forEach(s => {
      const m = String(s.species_id || '').trim().match(/^SPX(\d+)$/i);
      if (m) {
        const n = parseInt(m[1], 10);
        if (n > maxN) maxN = n;
      }
    });
    const speciesId = 'SPX' + String(maxN + 1).padStart(3, '0');
    const today = todayIso_();

    const spHeaders = shSpecies.getLastRow() >= 1
      ? shSpecies.getRange(1, 1, 1, shSpecies.getLastColumn()).getValues()[0].map(h => String(h).trim())
      : [];
    const spMap = {
      species_id: speciesId,
      species_name_th: name,
      water_type: waterType,
      pricing_unit: pricingUnit,
      reference_price: refPrice,
      price_updated_at: today,
      is_active: true
    };
    if (spHeaders.length >= 5) {
      shSpecies.appendRow(spHeaders.map(h => Object.prototype.hasOwnProperty.call(spMap, h) ? spMap[h] : ''));
    } else {
      shSpecies.appendRow([speciesId, name, waterType, pricingUnit, refPrice, today, true]);
    }

    if (requireWeight) {
      const shRules = ss.getSheetByName(CONFIG.SHEETS.RULES);
      if (shRules) {
        const existingRules = readObjects_(shRules);
        const already = existingRules.some(r => String(r.species_id || '').trim() === speciesId);
        if (!already) {
          const ruleHeaders = shRules.getLastRow() >= 1
            ? shRules.getRange(1, 1, 1, shRules.getLastColumn()).getValues()[0].map(h => String(h).trim())
            : [];
          const ruleMap = {
            species_id: speciesId,
            species_name_th: name,
            require_weight_on_death: true,
            note: 'เพิ่มนอกบัญชีกลาง',
            rule_note: 'เพิ่มนอกบัญชีกลาง'
          };
          if (ruleHeaders.length >= 2) {
            shRules.appendRow(ruleHeaders.map(h => Object.prototype.hasOwnProperty.call(ruleMap, h) ? ruleMap[h] : ''));
          } else {
            shRules.appendRow([speciesId, name, true, 'เพิ่มนอกบัญชีกลาง']);
          }
        }
      }
    }

    return {
      species_id: speciesId,
      species_name_th: name,
      water_type: waterType,
      pricing_unit: pricingUnit,
      reference_price: refPrice,
      price_updated_at: today,
      is_active: true,
      require_weight_on_death: requireWeight
    };
  });
}

/**
 * ซิงก์ชนิดสัตว์น้ำและราคากลางจาก "ชีตกลาง" (Central Master Sheet) ในอนาคต
 * โดยไม่กระทบราคา Snapshot ในล็อตเดิมที่รับเข้าแล้ว และรักษาชนิดนอกบัญชีกลาง (SPX*) ไว้
 */
function syncCentralSpecies() {
  if (!CONFIG.CENTRAL_SPECIES_SHEET_ID) {
    throw new Error('ยังไม่ได้ตั้งค่า CENTRAL_SPECIES_SHEET_ID ใน CONFIG');
  }
  return withLock_(() => {
    const centralSS = SpreadsheetApp.openById(CONFIG.CENTRAL_SPECIES_SHEET_ID);
    const srcSheet = centralSS.getSheetByName(CONFIG.CENTRAL_SPECIES_TAB);
    const values = srcSheet.getDataRange().getValues();
    if (values.length < 2) throw new Error('ชีตกลางไม่มีข้อมูล');

    const dstSheet = getSS_().getSheetByName(CONFIG.SHEETS.SPECIES);
    const centralHeaders = values[0].map(h => String(h).trim());
    const centralIdIdx = centralHeaders.indexOf('species_id');
    const centralIds = {};
    for (let r = 1; r < values.length; r++) {
      const id = centralIdIdx !== -1 ? String(values[r][centralIdIdx] || '').trim() : '';
      if (id) centralIds[id] = true;
    }

    // Preserve custom non-central species (SPX*) before overwriting Master_Species
    const existingCustom = readObjects_(dstSheet).filter(obj => {
      const id = String(obj.species_id || '').trim();
      return /^SPX/i.test(id) && !centralIds[id];
    });
    const preservedRows = existingCustom.map(obj =>
      centralHeaders.map(h => Object.prototype.hasOwnProperty.call(obj, h) ? obj[h] : '')
    );

    dstSheet.clearContents();
    dstSheet.getRange(1, 1, values.length, values[0].length).setValues(values);
    if (preservedRows.length > 0) {
      dstSheet.getRange(values.length + 1, 1, preservedRows.length, centralHeaders.length).setValues(preservedRows);
    }
    return {
      synced_rows: values.length - 1,
      preserved_custom: preservedRows.length,
      synced_at: nowIso_()
    };
  });
}

function deductFifoLots_(ss, tankId, speciesId, qty, weightG, targetLotId, eventDate) {
  const shLots = ss.getSheetByName(CONFIG.SHEETS.LOTS);
  const rows = shLots.getDataRange().getValues();
  const idx = indexMap_(rows[0]);

  const candidates = [];
  for (let r = 1; r < rows.length; r++) {
    if (
      String(rows[r][idx.tank_id]).trim() === tankId &&
      String(rows[r][idx.species_id]).trim() === speciesId &&
      String(rows[r][idx.status]).trim().toUpperCase() === 'ACTIVE' &&
      Number(rows[r][idx.qty_remaining]) > 0
    ) {
      candidates.push({
        rowNum: r + 1,
        lot_id: String(rows[r][idx.lot_id]).trim(),
        received_date: formatDateCell_(rows[r][idx.received_date]),
        source_type: String(rows[r][idx.source_type]).trim(),
        pricing_unit: String(rows[r][idx.pricing_unit] || 'PER_UNIT').trim(),
        unit_cost: Number(rows[r][idx.unit_cost] || 0),
        warranty_until: idx.warranty_until != null ? formatDateCell_(rows[r][idx.warranty_until]) : '',
        qty_remaining: Number(rows[r][idx.qty_remaining])
      });
    }
  }

  candidates.sort((a, b) => a.received_date.localeCompare(b.received_date) || a.lot_id.localeCompare(b.lot_id));
  if (targetLotId && targetLotId !== 'AUTO_FIFO') {
    candidates.sort((a, b) => (a.lot_id === targetLotId ? -1 : b.lot_id === targetLotId ? 1 : 0));
  }

  let need = qty;
  const breakdown = [];
  let totalCost = 0;
  let hasWarranty = false;

  for (let i = 0; i < candidates.length && need > 0; i++) {
    const c = candidates[i];
    const take = Math.min(need, c.qty_remaining);
    const underWarranty = !!(c.warranty_until && eventDate && eventDate <= c.warranty_until);
    if (underWarranty) hasWarranty = true;
    let cost = 0;
    if (!underWarranty) {
      if (c.pricing_unit === 'PER_KG') {
        const kg = (Number(weightG) || 0) / 1000;
        cost = Math.round(kg * (take / qty) * c.unit_cost * 100) / 100;
      } else {
        cost = take * c.unit_cost;
      }
    }
    breakdown.push({
      rowNum: c.rowNum,
      lot_id: c.lot_id,
      received_date: c.received_date,
      source_type: c.source_type,
      pricing_unit: c.pricing_unit,
      unit_cost: c.unit_cost,
      warranty_until: c.warranty_until,
      under_warranty: underWarranty,
      qty_deducted: take,
      new_remaining: c.qty_remaining - take,
      cost: cost
    });
    totalCost += cost;
    need -= take;
  }

  if (need > 0) {
    throw new Error('สต็อกคงเหลือในตู้ ' + tankId + ' ไม่เพียงพอ (ขาดอีก ' + need + ' ตัว)');
  }

  breakdown.forEach(b => {
    shLots.getRange(b.rowNum, idx.qty_remaining + 1).setValue(b.new_remaining);
    if (b.new_remaining === 0) {
      shLots.getRange(b.rowNum, idx.status + 1).setValue('DEPLETED');
    }
  });

  return {
    totalCost: Math.round(totalCost * 100) / 100,
    hasWarranty: hasWarranty,
    breakdown: breakdown.map(b => ({
      lot_id: b.lot_id,
      received_date: b.received_date,
      source_type: b.source_type,
      pricing_unit: b.pricing_unit,
      unit_cost: b.unit_cost,
      warranty_until: b.warranty_until,
      under_warranty: b.under_warranty,
      qty_deducted: b.qty_deducted,
      cost: b.cost
    }))
  };
}

function assertMonthNotLocked_(ss, dateStr) {
  const ymTh = toBuddhistYM_(dateStr);
  const ymIso = toIsoYM_(dateStr.slice(0, 7));
  const snaps = readObjects_(ss.getSheetByName(CONFIG.SHEETS.SNAPSHOTS));
  const locked = snaps.some(s =>
    toBool_(s.is_locked) && (String(s.period_ym) === ymTh || String(s.period_ym) === ymIso)
  );
  if (locked) {
    throw new Error('เดือน ' + ymTh + ' ถูกปิดยอดและล็อกแล้ว ไม่สามารถบันทึกหรือแก้ไขย้อนหลังได้');
  }
}

let _lockDepth = 0;
function withLock_(fn) {
  if (_lockDepth > 0) {
    return fn();
  }
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  _lockDepth++;
  try {
    return fn();
  } finally {
    _lockDepth--;
    lock.releaseLock();
  }
}

function getSS_() {
  return SpreadsheetApp.openById(CONFIG.SPREADSHEET_ID);
}

function readObjects_(sheet) {
  if (!sheet || sheet.getLastRow() < 2) return [];
  const vals = sheet.getDataRange().getValues();
  const headers = vals[0].map(h => String(h).trim());
  const out = [];
  for (let r = 1; r < vals.length; r++) {
    const obj = {};
    let hasData = false;
    for (let c = 0; c < headers.length; c++) {
      obj[headers[c]] = vals[r][c];
      if (vals[r][c] !== '' && vals[r][c] !== null) hasData = true;
    }
    if (hasData) out.push(obj);
  }
  return out;
}

function indexMap_(headers) {
  const m = {};
  headers.forEach((h, i) => { m[String(h).trim()] = i; });
  return m;
}

function findSpecies_(ss, speciesId) {
  const list = readObjects_(ss.getSheetByName(CONFIG.SHEETS.SPECIES));
  return list.find(s => String(s.species_id) === String(speciesId)) || null;
}

function findTank_(ss, tankId) {
  const list = readObjects_(ss.getSheetByName(CONFIG.SHEETS.TANKS));
  return list.find(t => String(t.tank_id) === String(tankId)) || null;
}

function parseCarriedLots_(noteRaw, fallbackDate, fallbackUnit, fallbackCost, fallbackQty) {
  const marker = 'LOTS:';
  const pos = noteRaw.indexOf(marker);
  if (pos !== -1) {
    try {
      return JSON.parse(noteRaw.slice(pos + marker.length).trim());
    } catch (e) {}
  }
  return [{
    lot_id: '',
    received_date: fallbackDate,
    source_type: 'ตู้จัดแสดง',
    pricing_unit: fallbackUnit || 'PER_UNIT',
    unit_cost: fallbackCost || 0,
    qty_deducted: fallbackQty || 0
  }];
}

function toBool_(v) {
  return v === true || String(v).toUpperCase() === 'TRUE';
}

function formatDateCell_(v) {
  if (!v) return '';
  if (Object.prototype.toString.call(v) === '[object Date]') {
    return Utilities.formatDate(v, CONFIG.TIMEZONE, 'yyyy-MM-dd');
  }
  return String(v).trim();
}

function todayIso_() {
  return Utilities.formatDate(new Date(), CONFIG.TIMEZONE, 'yyyy-MM-dd');
}

function nowIso_() {
  return Utilities.formatDate(new Date(), CONFIG.TIMEZONE, "yyyy-MM-dd'T'HH:mm:ss'+07:00'");
}

function toBuddhistYM_(isoDate) {
  const parts = String(isoDate || '').split('-');
  if (parts.length < 2) return String(isoDate || '');
  let y = parseInt(parts[0], 10);
  if (y < 2400) y += 543;
  return y + '-' + parts[1];
}

function toIsoYM_(ymStr) {
  const parts = String(ymStr || '').split('-');
  if (parts.length < 2) return String(ymStr || '');
  let y = parseInt(parts[0], 10);
  if (y > 2400) y -= 543;
  return y + '-' + parts[1];
}

function nextSeqForPrefix_(sheet, idHeader, idPrefix) {
  if (!sheet || sheet.getLastRow() < 2) return 1;
  const rows = sheet.getDataRange().getValues();
  const headers = rows[0].map(h => String(h).trim());
  let col = idHeader ? headers.indexOf(idHeader) : -1;
  if (col === -1) col = 0;
  let maxSeq = 0;
  for (let r = 1; r < rows.length; r++) {
    const val = String(rows[r][col] || '').trim();
    if (idPrefix && val.indexOf(idPrefix) !== 0) continue;
    const m = val.match(/-(\d+)$/);
    if (m) {
      const n = parseInt(m[1], 10);
      if (n > maxSeq) maxSeq = n;
    }
  }
  return maxSeq + 1;
}

function makeId_(prefix, isoDate, seqOrSheet, idHeader) {
  const ym = toBuddhistYM_(isoDate).replace('-', '');
  const fullPrefix = prefix + '-' + ym + '-';
  const seq = (typeof seqOrSheet === 'number')
    ? seqOrSheet
    : nextSeqForPrefix_(seqOrSheet, idHeader || '', fullPrefix);
  return fullPrefix + String(seq).padStart(3, '0');
}
